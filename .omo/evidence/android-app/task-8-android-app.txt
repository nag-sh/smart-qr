Task 8: Add GitHub Actions debug APK workflow

Workflow file: .github/workflows/android-pr.yml

Notes:
- Triggers on push to main and pull_request to main.
- Builds an unsigned debug APK using ./gradlew assembleDebug (no release signing).
- Uploads the produced APK as an artifact named app-debug.apk.
- Artifact path glob: **/android/app/build/outputs/apk/debug/*.apk

--- workflow contents ---

name: Android PR Debug APK

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build-debug-apk:
    name: Build Android Debug APK
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: client/package-lock.json

      - name: Install root dependencies
        run: npm ci

      - name: Install client dependencies
        working-directory: client
        run: npm ci

      - name: Build mobile web assets
        working-directory: client
        run: npm run build:mobile

      - name: Sync Capacitor Android project
        working-directory: client
        run: npx cap sync android

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Set up Android SDK
        uses: android-actions/setup-android@v3
        with:
          accept-android-licenses: true

      - name: Assemble debug APK
        working-directory: client/android
        run: chmod +x ./gradlew && ./gradlew assembleDebug

      - name: Upload debug APK
        uses: actions/upload-artifact@v4
        with:
          name: app-debug.apk
          path: '**/android/app/build/outputs/apk/debug/*.apk'
